cmake_minimum_required(VERSION 3.10)
project(libgurt C)
set(CMAKE_C_STANDARD 99)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(GURT_SRC src/despeckle.c src/deglitch.c src/gapfill.c src/unfold.c src/tormodel.c)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/axisym.c)   # full GURT build only
  list(APPEND GURT_SRC src/axisym.c)
endif()
add_library(gurt SHARED ${GURT_SRC})
target_include_directories(gurt PUBLIC include)
set_target_properties(gurt PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_library(M_LIB m)
if(M_LIB)
  target_link_libraries(gurt PRIVATE ${M_LIB})
endif()
add_library(gurt_static STATIC ${GURT_SRC})
target_include_directories(gurt_static PUBLIC include)
set_target_properties(gurt_static PROPERTIES OUTPUT_NAME gurt POSITION_INDEPENDENT_CODE ON)
